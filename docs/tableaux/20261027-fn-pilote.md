### `b_spread` — bandes ponctuelle et simultanée du lissage `lowess` sur `z_boot`, J1 et J2, in-sample (§ 1.2)

```r
setwd("/home/user/outil_usp")
for (n in c("premium","premium_interieur")) {
  r <- readRDS(file.path("tests/reference", paste0(n, ".rds")))
  pd <- r$plots_data; x <- pd$spread$x; zb <- r$bootstrap$z_boot
  T <- length(x)
  cat("=====", n, " x =", paste(x, collapse=" "), "\n")
  cat("doublons x :", any(duplicated(x)), "\n")
  lo_obs <- lowess(x, sqrt(abs(r$plots_data$residus$z)))
  stopifnot(isTRUE(all.equal(lo_obs$y, pd$spread_lisse$y)))
  garde <- apply(is.finite(zb),1,all); Beff <- sum(garde)
  L <- t(apply(zb[garde,,drop=FALSE], 1, function(z) lowess(x, sqrt(abs(z)))$y))
  # lo$x is sort(x)
  q <- apply(L, 2, quantile, probs=c(.05,.95), type=7)
  cat("Beff =", Beff, "\n")
  cat("bande bas  :", sprintf("%.3f", q[1,]), "\n")
  cat("courbe obs :", sprintf("%.3f", lo_obs$y), "\n")
  cat("bande haut :", sprintf("%.3f", q[2,]), "\n")
  hors <- function(b,h) mean(rowSums(L < rep(b,each=nrow(L)) | L > rep(h,each=nrow(L)))>0)
  cat("taux global de sortie (in-sample) bande ponctuelle 90 % :", hors(q[1,],q[2,]), "\n")
  cat("courbe observee hors bande ponctuelle en au moins un point :", any(lo_obs$y < q[1,] | lo_obs$y > q[2,]), "\n")
  Sc <- apply(L, 2, sort); kst <- NA
  for (k in seq_len(Beff %/% 2)) { tk <- hors(Sc[k,], Sc[Beff+1-k,]); if (tk > 0.10) break; kst <- k; tkk <- tk }
  cat("k* =", kst, " taux =", tkk, "\n")
  cat("sim bas :", sprintf("%.3f", Sc[kst,]), "\n sim haut:", sprintf("%.3f", Sc[Beff+1-kst,]), "\n")
  cat("courbe obs hors bande simultanee :", any(lo_obs$y < Sc[kst,] | lo_obs$y > Sc[Beff+1-kst,]), "\n")
  # pente fin - debut et amplitude sous le modele
  pente <- abs(L[,T]-L[,1]); amp <- apply(L,1,function(v) diff(range(v)))
  cat(sprintf("|fin-debut| > 0.3 : %.3f ; amplitude mediane %.3f q90 %.3f\n", mean(pente>0.3), median(amp), quantile(amp,.9)))
  cat(sprintf("obs : |fin-debut| = %.3f, amplitude = %.3f ; rang de l'amplitude obs = %.3f\n", abs(lo_obs$y[T]-lo_obs$y[1]), diff(range(lo_obs$y)), mean(amp >= diff(range(lo_obs$y)))))
  # regimes
  d <- r$bootstrap$delta_boot
  reg <- ifelse(d <= 1e-6, "bord0", ifelse(d >= 1-1e-6, "bord1", "int"))[garde]
  for (g in unique(reg)) { qg <- apply(L[reg==g,,drop=FALSE],2,quantile,probs=c(.05,.95)); cat(g, sum(reg==g), " bas:", sprintf("%.3f", qg[1,]), "\n          haut:", sprintf("%.3f", qg[2,]), "\n") }
}
```

### `c_cout` — coûts unitaires d'`usp_bootstrap()` et d'`usp_ajuster()` ; identité de la boucle allégée avec `usp_bootstrap()` (§ 2.7)

```r
setwd("/home/user/outil_usp")
source("R/engine.R")
ln <- read.csv("tests/donnees/donnees_ln.csv"); li <- read.csv("tests/donnees/donnees_ln_interieur.csv")
leger <- function(fit, B, seed) engine_sous_graine(seed, {
  sig <- sig_r <- del <- rep(NA_real_, B)
  for (b in seq_len(B)) {
    yb <- usp_simuler(fit)
    f <- try(usp_ajuster_rapide(fit$x, yb, fit$delta, fit$gamma), silent = TRUE)
    if (inherits(f, "try-error")) next
    sig[b] <- f$sigma; del[b] <- f$delta
    fr <- try(usp_ajuster_contraint(fit$x, yb, fit$delta, fit$gamma), silent = TRUE)
    if (!inherits(fr, "try-error")) sig_r[b] <- fr$sigma
  }
```
