# Comparaisons appariees de #307 (Q-D1-Fj-6) sur le brut versionne de #309.
# Regle de decision de #309 : rejet si p_mc < alpha ; p absente = non-rejet.
f <- "docs/tableaux/20261021-issue309-puissance-mw-brut.tsv"
md5 <- unname(tools::md5sum(f))
D <- utils::read.delim(f, check.names = FALSE, stringsAsFactors = FALSE, na.strings = "NA")
D <- D[D$type == "rep" & D$statut == "admis" & startsWith(D$cle, "B-"), ]
dec <- function(x, a) { p <- x; is.finite(p) & p < a }
mcn <- function(d1, d2) {
  n01 <- sum(d1 & !d2); n10 <- sum(!d1 & d2); n <- n01 + n10
  c(n01 = n01, n10 = n10, p = if (n) stats::binom.test(n01, n, 0.5)$p.value else 1)
}
PAIRES <- list(c("PenteRel", "PenteIntra"), c("PenteRel", "Origine"), c("PenteRel", "Alpha"), c("PenteIntra", "Origine"))
fp <- function(p) if (p < 1e-3) formatC(p, format = "e", digits = 1, decimal.mark = ",") else formatC(p, format = "f", digits = 3, decimal.mark = ",")
fr <- function(x) formatC(x, format = "f", digits = 3, decimal.mark = ",")
cles <- unique(D$cle)
lig <- character(0)
for (pr in PAIRES) for (cl in cles) for (a in c(0.10, 0.05)) {
  d <- D[D$cle == cl, ]
  d1 <- dec(d[[paste0("p:", pr[1])]], a); d2 <- dec(d[[paste0("p:", pr[2])]], a)
  m <- mcn(d1, d2)
  lig <- c(lig, paste("|", pr[1], "/", pr[2], "|", sub("^B-", "", cl), "|", nrow(d), "|",
                      formatC(a, format = "f", digits = 2, decimal.mark = ","), "|", fr(mean(d1)), "|", fr(mean(d2)), "|",
                      sprintf("%d / %d", m[["n01"]], m[["n10"]]), "|", fp(m[["p"]]), "|", fr(mean(d1 & d2)), "|"))
}
cat("md5 du brut :", md5, "\n")
cat("| Paire (X / Y) | Cellule | n | seuil | taux X | taux Y | n01 / n10 | McNemar exact bilatéral | rejet conjoint |\n|---|---|---|---|---|---|---|---|---|\n")
cat(lig, sep = "\n")
cat("\n| Cellule H0 | n | corrélation de rang (Spearman) des p_mc, PenteRel / Alpha | PenteRel / PenteIntra |\n|---|---|---|---|\n")
for (cl in grep("\\.H0$", cles, value = TRUE)) {
  d <- D[D$cle == cl, ]
  r1 <- stats::cor(d[["p:PenteRel"]], d[["p:Alpha"]], method = "spearman", use = "complete.obs")
  r2 <- stats::cor(d[["p:PenteRel"]], d[["p:PenteIntra"]], method = "spearman", use = "complete.obs")
  cat("|", sub("^B-", "", cl), "|", nrow(d), "|", formatC(r1, format = "f", digits = 3, decimal.mark = ","), "|", formatC(r2, format = "f", digits = 3, decimal.mark = ","), "|\n")
}
